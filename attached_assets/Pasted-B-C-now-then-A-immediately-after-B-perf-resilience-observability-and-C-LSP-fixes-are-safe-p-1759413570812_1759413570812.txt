Weather/network calls: timeout + retry + cache
Add a tiny fetchSafe and in-memory cache so user actions never stall.

// server/lib/fetchSafe.ts
export async function fetchSafe(url: string, opts: RequestInit = {}, tries = 3, timeoutMs = 4000) {
  for (let i = 0; i < tries; i++) {
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), timeoutMs);
    try {
      const res = await fetch(url, { ...opts, signal: ctrl.signal });
      clearTimeout(timer);
      if (res.ok) return res;
      if (res.status >= 500) throw new Error("retryable");
      return res;
    } catch {
      clearTimeout(timer);
      if (i === tries - 1) throw new Error("fetch failed");
      await new Promise(r => setTimeout(r, 250 * (i + 1) * (i + 1)));
    }
  }
  throw new Error("unreachable");
}

// server/weatherService.ts (example)
const CACHE = new Map<string, { at: number; data: any }>();
const DAY = 86_400_000;
export async function getWeather(key: string) {
  const c = CACHE.get(key);
  if (c && Date.now() - c.at < DAY) return c.data;
  const res = await fetchSafe(makeUrl(key), {}, 3, 4000).catch(() => null);
  if (!res || !res.ok) return null;
  const data = await res.json();
  CACHE.set(key, { at: Date.now(), data });
  return data;
}

import { Component, ReactNode } from "react";
export default class ErrorBoundary extends Component<{children:ReactNode},{hasError:boolean}> {
  state = { hasError: false };
  static getDerivedStateFromError() { return { hasError: true }; }
  componentDidCatch(err:any, info:any){ console.error("UI Error", err, info); }
  render(){ return this.state.hasError ? <div className="p-6">Something went wrong. Please refresh.</div> : this.props.children; }
}
