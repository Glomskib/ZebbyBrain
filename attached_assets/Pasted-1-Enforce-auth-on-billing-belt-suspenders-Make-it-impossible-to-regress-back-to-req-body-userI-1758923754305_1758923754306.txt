1) Enforce auth on billing (belt & suspenders)

Make it impossible to regress back to req.body.userId.

// server/src/middleware/requireAuth.ts
import { Request, Response, NextFunction } from "express";
export function requireAuth(req: Request, res: Response, next: NextFunction) {
  const user = (req as any).user;
  if (!user?.id) return res.status(401).json({ error: "auth_required" });
  next();
}


Apply to both endpoints:

// server/src/index.ts
import { requireAuth } from "./middleware/requireAuth";
app.use("/api/billing", requireAuth, billing);
app.use("/api/billing-portal", requireAuth, billingPortal); // if separate


And in billing.ts, delete any userId from body/params; only use req.user.id.

2) Unit tests that prevent future regressions

Add these Supertest checks:

// server/tests/billing.auth.test.ts
import request from "supertest";
import app from "../src/index"; // your express app

describe("Billing auth", () => {
  test("rejects unauthenticated checkout session", async () => {
    const res = await request(app)
      .post("/api/billing/create-checkout-session")
      .send({ userId: "forged" }); // should be ignored anyway
    expect(res.status).toBe(401);
  });

  test("rejects forged userId even if token missing", async () => {
    const res = await request(app)
      .post("/api/billing/create-checkout-session")
      .send({ userId: "someone-else" });
    expect(res.status).toBe(401);
  });
});

3) Promo/trial abuse controls (lightweight, effective)

Prevent edge-case abuse of coupons/trials.

const MAX_TRIAL_MONTHS = 3;

  return res.status(400).json({ error: "invalid_trial" });

// Optional: whitelist coupons you’ll accept

// when couponId provided:

  return res.status(400).json({ error: "coupon_not_allowed" });

4) Idempotency + rate-limit on billing creation

Avoid duplicate subscriptions if a user double-clicks.

// Idempotency key based on user + minute
import crypto from "crypto";
const idemKey = crypto
  .createHash("sha256")
  .update(`${req.user.id}:${new Date().toISOString().slice(0,16)}`)
  .digest("hex");

const session = await stripe.checkout.sessions.create(params, {
  idempotencyKey: idemKey
});


And throttle:

import rateLimit from "express-rate-limit";
app.use("/api/billing/create-checkout-session",
  rateLimit({ windowMs: 60_000, max: 5 })
);
