2) Import Settings page (Zebby-styled)

src/pages/ImportSettings.tsx

import { useState } from 'react';

type SourceState = {
  googleFit: boolean;
  appleHealth: boolean;
  lastSync?: string;
  status?: 'idle'|'syncing'|'ok'|'warn';
}

export default function ImportSettings(){
  const [s, setS] = useState<SourceState>({
    googleFit: true, appleHealth: false, lastSync: '2025-09-24 21:10', status: 'ok'
  });
  const [loading, setLoading] = useState(false);

  async function syncNow(){
    setLoading(true);
    try {
      // call your /api/import/sync endpoint
      await fetch('/api/import/sync', { method: 'POST' });
      setS(prev => ({ ...prev, lastSync: new Date().toISOString().slice(0,16).replace('T',' '), status: 'ok' }));
    } catch {
      setS(prev => ({ ...prev, status: 'warn' }));
    } finally { setLoading(false); }
  }

  return (
    <div className="max-w-6xl mx-auto px-4 pb-12">
      <div className="appbar"><div className="appbar-inner">
        <div className="flex items-center gap-2">
          <div className="h-8 w-8 rounded-full grid place-items-center bg-zebby-amber text-zebby-slate">🦓</div>
          <div className="font-display text-xl">Zebby — Health Data Import</div>
        </div>
        <button onClick={syncNow} className="btn btn-primary disabled:opacity-60" disabled={loading}>
          {loading ? 'Syncing…' : 'Sync now'}
        </button>
      </div></div>

      {/* Backdrop banner */}
      <div className="mt-6 rounded-2xl skyback p-1">
        <div className="tile bg-white/80 dark:bg-white/10">
          <div className="section-h">
            <div className="section-t">Sources</div>
            <div className="flex gap-2">
              <span className={`badge ${s.status==='ok'?'badge-ok':s.status==='warn'?'badge-warn':'badge-info'}`}>
                {s.status==='ok' && 'Up to date'}
                {s.status==='warn' && 'Needs attention'}
                {!s.status && 'Idle'}
              </span>
              {s.lastSync && <span className="badge badge-info">Last sync: {s.lastSync}</span>}
            </div>
          </div>

          {/* Sources grid */}
          <div className="grid md:grid-cols-2 gap-4">
            <SourceCard
              icon="🟢" title="Google Fit"
              enabled={s.googleFit}
              onToggle={(v)=>setS(p=>({...p, googleFit:v}))}
              description="Import heart rate, steps & sleep. Toggle on to include in Zebby."
            />
            <SourceCard
              icon="🟣" title="Apple Health"
              enabled={s.appleHealth}
              onToggle={(v)=>setS(p=>({...p, appleHealth:v}))}
              description="Import HR, sleep, and activity summaries (via your device export)."
            />
          </div>
        </div>
      </div>

      {/* Privacy & scope */}
      <div className="mt-6 grid md:grid-cols-3 gap-4">
        <div className="tile">
          <div className="section-t text-lg">Privacy-first</div>
          <p className="opacity-80 text-sm">You control sources. Imports run locally on your account; Zebby doesn’t share data.</p>
        </div>
        <div className="tile">
          <div className="section-t text-lg">Conflict rules</div>
          <p className="opacity-80 text-sm">We avoid duplicates, and newer records win unless you approve otherwise.</p>
        </div>
        <div className="tile">
          <div className="section-t text-lg">Manual review</div>
          <p className="opacity-80 text-sm">Potential conflicts go to a queue for your approval (see below).</p>
        </div>
      </div>

      <ReviewQueuePreview />
    </div>
  );
}

function SourceCard({
  icon, title, description, enabled, onToggle
}: { icon:string; title:string; description:string; enabled:boolean; onToggle:(v:boolean)=>void }){
  return (
    <div className="tile">
      <div className="flex items-center justify-between">
        <div className="flex items-center gap-3">
          <div className="h-12 w-12 rounded-2xl grid place-items-center bg-white/80 dark:bg-white/10 text-2xl">{icon}</div>
          <div>
            <div className="font-display text-xl">{title}</div>
            <div className="text-sm opacity-80">{description}</div>
          </div>
        </div>
        <button
          onClick={()=>onToggle(!enabled)}
          className={`switch ${enabled ? '' : 'switch-off'}`}
          aria-pressed={enabled}
        >
          <span className={`knob ${enabled ? 'knob-on' : ''}`} />
        </button>
      </div>
    </div>
  );
}

3) Review Queue page (Zebby layout + colors)

src/pages/ImportReviewQueue.tsx

import { useState, useEffect } from 'react';

type Conflict = {
  id: string;
  type: 'duplicate'|'conflict';
  metric: string; // "HR", "Sleep", etc.
  incoming: { source:'Google Fit'|'Apple Health'; value:string; time:string };
  existing?: { value:string; time:string };
};

export default function ImportReviewQueue(){
  const [items, setItems] = useState<Conflict[]>([]);

  useEffect(()=>{
    // fetch('/api/import/conflicts').then(r=>r.json()).then(setItems);
    setItems([
      { id:'1', type:'conflict', metric:'HR (standing)', incoming:{source:'Google Fit', value:'104 bpm', time:'10:42'}, existing:{value:'100 bpm', time:'10:43'} },
      { id:'2', type:'duplicate', metric:'Sleep', incoming:{source:'Apple Health', value:'7h 12m', time:'2025-09-24'}, existing:{value:'7h 10m', time:'2025-09-24'} },
    ]);
  },[]);

  async function resolve(id:string, action:'keep_existing'|'replace'|'merge'|'drop'){
    // POST /api/import/conflicts/:id/resolve { action }
    setItems(prev => prev.filter(x=>x.id!==id));
  }

  return (
    <div className="max-w-6xl mx-auto px-4 pb-14">
      <div className="appbar"><div className="appbar-inner">
        <div className="flex items-center gap-2">
          <div className="h-8 w-8 rounded-full grid place-items-center bg-zebby-amber text-zebby-slate">🦓</div>
          <div className="font-display text-xl">Zebby — Review queue</div>
        </div>
      </div></div>

      <div className="mt-6 rounded-2xl skyback p-1">
        <div className="tile bg-white/85 dark:bg-white/10">
          {items.length === 0 ? (
            <div className="text-center py-12">
              <div className="text-5xl mb-3">🎉</div>
              <div className="font-semibold">No pending items</div>
              <div className="opacity-70 text-sm">You’re all caught up.</div>
            </div>
          ) : (
            <div className="space-y-3">
              {items.map(it => (
                <div key={it.id} className="row">
                  <div className="col-span-2">
                    <span className={`badge ${it.type==='conflict' ? 'badge-warn' : 'badge-info'}`}>
                      {it.type==='conflict' ? 'Conflict' : 'Duplicate?'}
                    </span>
                  </div>
                  <div className="col-span-6">
                    <div className="font-semibold">{it.metric}</div>
                    <div className="text-xs opacity-70">
                      Incoming ({it.incoming.source}) {it.incoming.value} @ {it.incoming.time}
                      {it.existing && <> • Existing {it.existing.value} @ {it.existing.time}</>}
                    </div>
                  </div>
                  <div className="col-span-4 flex justify-end gap-2">
                    {it.type==='conflict' ? (
                      <>
                        <button onClick={()=>resolve(it.id,'keep_existing')} className="btn btn-ghost">Keep existing</button>
                        <button onClick={()=>resolve(it.id,'replace')} className="btn btn-primary">Replace</button>
                      </>
                    ) : (
                      <>
                        <button onClick={()=>resolve(it.id,'merge')} className="btn btn-ghost">Merge</button>
                        <button onClick={()=>resolve(it.id,'drop')} className="btn btn-primary">Keep one</button>
                      </>
                    )}
                  </div>
                </div>
              ))}
              <div className="text-sm opacity-70">Total: {items.length} items</div>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}

4) Where to plug these in

Add routes:

/import/settings → ImportSettings

/import/review → ImportReviewQueue

Use the skyback wrapper and tile/card/badge/btn classes anywhere you want the Zebby vibe.

Replace any old gray buttons with .btn + .btn-primary, and wrap critical info in .badge-*.
