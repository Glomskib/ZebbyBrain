1) HTTP security headers (server-wide)
// server/src/app.ts (or index.ts)
import helmet from "helmet";
import cors from "cors";
import express from "express";

const app = express();

// Strict CORS: only your known origins
const allowed = [process.env.APP_URL, "http://localhost:5173"].filter(Boolean) as string[];
app.use(cors({ origin: allowed, credentials: true }));

app.use(helmet({
  contentSecurityPolicy: {
    useDefaults: true,
    directives: {
      "default-src": ["'self'"],
      "img-src": ["'self'", "data:", "blob:"],
      "script-src": ["'self'"],
      "style-src": ["'self'", "'unsafe-inline'"], // if using inline Tailwind injection
      "connect-src": ["'self'", "https://api.stripe.com"],
      "frame-ancestors": ["'none'"],
    },
  },
  referrerPolicy: { policy: "no-referrer" },
  crossOriginEmbedderPolicy: false, // if you use pdf rendering in-browser
}));


Why: shuts down a big class of drive-by attacks and enforces strict origins.

4) Request schema guards (centralize)

You’ve added Zod per-route. Add a tiny helper to eliminate repeats:

// server/src/lib/validate.ts
import { ZodSchema } from "zod";
export const validate =
  (schema: ZodSchema) =>
  (req, res, next) => {
    const parsed = schema.safeParse({ body: req.body, query: req.query, params: req.params });
    if (!parsed.success) return res.status(400).json({ error: "validation", detail: parsed.error.flatten() });
    Object.assign(req, parsed.data); // optional: typed attach
    next();
  };


Use app.post("/api/water-intake", validate(schema), handler).
