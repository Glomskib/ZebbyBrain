Server logic:

function resolveRange(userId, window, tz) {
  const now = new Date();
  switch(window) {
    case "today":  return { from: startOfTodayInTZ(tz), to: now };
    case "up_to_now": return { from: accountCreatedAt(userId), to: now };
    case "last_symptom": {
      const last = latestSymptomOccurredAt(userId);
      return { from: last ?? now, to: now }; // if none, fall back gracefully
    }
    default: { // “Xd”
      const days = parseInt(window.replace("d",""), 10);
      const from = new Date(now); from.setDate(from.getDate() - days);
      return { from, to: now };
    }
  }
}
