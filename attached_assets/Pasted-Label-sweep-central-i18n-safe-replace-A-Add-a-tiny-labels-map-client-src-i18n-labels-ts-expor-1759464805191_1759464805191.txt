3) Ask Zebby polish (counter + upgrade wiring)

Your page already exists. Make sure it:

Reads X-Queries-Left from the response headers

Stores it in state and renders a counter

Shows an Upgrade CTA when left ≤ 0

Calls your Stripe checkout route and redirects to returned URL

Example (merge into client/src/pages/ask-zebby.tsx)

const [queriesLeft, setQueriesLeft] = useState<number | null>(null);
const [atLimit, setAtLimit] = useState(false);
const [loading, setLoading] = useState(false);
const [error, setError] = useState<string | null>(null);

async function ask(question: string) {
  if (!question.trim()) return;
  setLoading(true); setError(null);
  try {
    const res = await fetch("/api/rag/answer", {
      method: "POST",
      headers: { "Content-Type": "application/json" }, // Bearer handled server-side if using session; mobile sends Bearer
      credentials: "include",
      body: JSON.stringify({ question, window: "30d" }),
    });

    if (!res.ok) {
      if (res.status === 429) {
        setAtLimit(true);
        setError("Daily limit reached.");
      } else if (res.status === 401) {
        setError("Please log in to ask questions.");
      } else {
        setError(`Error: ${res.statusText}`);
      }
      return;
    }

    const left = res.headers.get("X-Queries-Left");
    if (left) {
      const n = parseInt(left, 10);
      setQueriesLeft(n);
      setAtLimit(n <= 0);
    }

    const data = await res.json();
    // set your `answer`, `sources` etc here...
  } finally {
    setLoading(false);
  }
}

async function upgrade() {
  const r = await fetch("/api/billing/create-checkout-session", {
    method: "POST",
    credentials: "include",
  });
  const { url } = await r.json();
  window.location.href = url;
}
