Create server/src/routes/aiInsights.ts

import { Router } from "express";
import rateLimit from "express-rate-limit";
import { FF } from "../config/flags";
import { getUserStats } from "../services/insightStats"; // build below
import NodeCache from "node-cache";

const cache = new NodeCache({ stdTTL: 60 * 60 * 24 }); // 24h
const r = Router();

r.use(rateLimit({ windowMs: 60_000, max: 10 }));

r.post("/insights", async (req, res) => {
  if (!FF.AI_TRENDS) return res.status(404).json({ error: "disabled" });
  const userId = (req as any).user?.id;
  if (!userId) return res.status(401).json({ error: "auth_required" });

  const range = (req.body?.range as "30d"|"90d") ?? "30d";
  const key = `ai:${userId}:${range}`;
  const cached = cache.get(key); if (cached) { res.set("Cache-Control","no-store"); return res.json(cached); }

  const facts = await getUserStats(userId, range); // numeric only, no PII

  // safety: truncate any inputs
  const sys = `You are Zebby, a supportive EDS companion. Provide non-medical-advice insights.
Use the numeric facts to identify trends, gentle ideas, and discussion prompts.
Never diagnose. Keep it brief, kind, and practical.`;

  const prompt = [
    { role: "system", content: sys },
    { role: "user", content:
      `Here are anonymized facts for ${range}.
Facts (JSON): ${JSON.stringify(facts)}
Please return JSON with keys: analysis (<=140 words), trendBullets (3-5 short bullets), actionIdeas (3 concise, optional steps).`}
  ];

  let result;
  try {
    const resp = await fetch("https://api.openai.com/v1/chat/completions",{
      method:"POST",
      headers:{ "Content-Type":"application/json", "Authorization":`Bearer ${process.env.OPENAI_API_KEY}`},
      body: JSON.stringify({ model:"gpt-4o-mini", messages: prompt, temperature:0.3 })
    });
    const data = await resp.json();
    const text = data.choices?.[0]?.message?.content ?? "{}";
    result = JSON.parse(text);
  } catch {
    result = { analysis:"We couldn’t generate insights right now.", trendBullets:[], actionIdeas:[] };
  }

  const payload = {
    range, analysis: String(result.analysis ?? ""),
    trendBullets: (result.trendBullets ?? []).slice(0,5).map(String),
    actionIdeas: (result.actionIdeas ?? []).slice(0,3).map(String),
    disclaimer: "Informational only — not medical advice."
  };

  cache.set(key, payload);
  res.set("Cache-Control","no-store");
  return res.json(payload);
});

export default r;

import { useEffect, useState } from "react";

export default function ZebbyFindings({ range = "30d" }: { range?: "30d"|"90d" }) {
  const [data, setData] = useState<{analysis:string;trendBullets:string[];actionIdeas:string[];disclaimer:string} | null>(null);
  const [loading, setLoading] = useState(false);

  useEffect(()=>{(async()=>{
    setLoading(true);
    try{
      const token = localStorage.getItem("zebby_token")||"";
      const r = await fetch("/api/ai/insights",{ method:"POST", headers:{ "Content-Type":"application/json", ...(token?{Authorization:`Bearer ${token}`}:{}) }, body: JSON.stringify({ range }) });
      const j = await r.json(); setData(j);
    } finally { setLoading(false); }
  })()},[range]);

  if (loading) return <div className="tile">Loading Zebby’s findings…</div>;
  if (!data) return <div className="tile">No insights yet.</div>;

  return (
    <div className="tile">
      <div className="font-display text-xl mb-1">Zebby Findings</div>
      <p className="opacity-80 mb-3">{data.analysis}</p>
      {!!data.trendBullets?.length && (
        <ul className="list-disc pl-5 mb-3">
          {data.trendBullets.map((b,i)=><li key={i}>{b}</li>)}
        </ul>
      )}
      {!!data.actionIdeas?.length && (
        <div className="mt-2">
          <div className="font-semibold mb-1">Tiny optional steps</div>
          <ul className="list-disc pl-5">
            {data.actionIdeas.map((a,i)=><li key={i}>{a}</li>)}
          </ul>
        </div>
      )}
      <div className="text-xs opacity-60 mt-3">{data.disclaimer}</div>
    </div>
  );
}

  const userId = (req as any).user?.id; if (!userId) return res.status(401).json({ error:"auth_required" });
  const { message, entryContext } = req.body ?? {};

  const facts = { /* pull quick snapshot like in getUserStats, short */ };
  const sys = `You are Zebby, a warm, validating EDS companion. You never diagnose.

    `User message: ${String(message||"")}`,

    const data = await resp.json();
    const text = data.choices?.[0]?.message?.content ?? "I’m here with you.";
    res.set("Cache-Control","no-store"); return res.json({ reply: text });
  }catch(e){
    return res.status(500).json({ error:"ai_unavailable" });
  }
});

Server proxy server/src/routes/shopify.ts

import { Router } from "express";
import rateLimit from "express-rate-limit";
import { FF } from "../config/flags";

const r = Router();
r.use(rateLimit({ windowMs: 60_000, max: 30 }));

async function gql(query:string, variables:any){
  const resp = await fetch(`https://${process.env.SHOPIFY_STORE_DOMAIN}/api/2024-07/graphql.json`,{
    method:"POST",
    headers:{
      "Content-Type":"application/json",
      "X-Shopify-Storefront-Access-Token": process.env.SHOPIFY_STOREFRONT_TOKEN||""
    },
    body: JSON.stringify({ query, variables })
  });
  return resp.json();
}

r.get("/products", async (req,res)=>{
  if (!FF.SHOPIFY) return res.status(404).json({ error:"disabled" });
  const { cursor, collection } = req.query as any;
  const query = `
    query($cursor:String, $handle:String){
      collectionByHandle(handle:$handle){ id title }
      products(first:20, after:$cursor, query:$handle){
        edges{ cursor node{ id handle title descriptionHtml featuredImage{url altText} priceRange{minVariantPrice{amount currencyCode}} } }
        pageInfo{ hasNextPage }
      }
    }`;
  const data = await gql(query, { cursor: cursor||null, handle: collection||null });
  res.set("Cache-Control","no-store"); res.json(data);
});

export default r;
