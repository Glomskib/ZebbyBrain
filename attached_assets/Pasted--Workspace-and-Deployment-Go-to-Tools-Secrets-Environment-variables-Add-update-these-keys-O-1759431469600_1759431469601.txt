B) Free vs paid request caps (server)

If you haven’t already, add a simple per-user daily quota (pseudo-code matches your existing middleware style):

// server/middleware/quota.ts
import type { RequestHandler } from "express";

const store = new Map<string, { date: string; count: number }>();

export const quota =
  (limitFree: number, limitPaid: number): RequestHandler =>
  (req, res, next) => {
    const user = (req as any).user as { id: string; plan?: "free" | "paid" } | undefined;
    if (!user) return res.status(401).json({ message: "Unauthorized" });

    const today = new Date().toISOString().slice(0, 10);
    const key = `${user.id}:${today}`;
    const rec = store.get(key) ?? { date: today, count: 0 };
    if (rec.date !== today) { rec.date = today; rec.count = 0; }

    const limit = user?.plan === "paid" ? Number(process.env.DAILY_QUERY_LIMIT_PAID ?? 100) 
                                        : Number(process.env.DAILY_QUERY_LIMIT_FREE ?? 5);

    if (rec.count >= limit) {
      return res.status(429).json({ message: "Daily limit reached" });
    }
    rec.count++;
    store.set(key, rec);
    next();
  };


Apply it to the costly endpoints:

// server/index.ts or routes
import { quota } from "./middleware/quota";
app.post("/api/rag/answer", isAuthenticated, quota(
  Number(process.env.DAILY_QUERY_LIMIT_FREE ?? 5),
  Number(process.env.DAILY_QUERY_LIMIT_PAID ?? 100)
), ragAnswerHandler);

C) Log approximate token usage per request

If you’re using OpenAI’s Node SDK, add a tiny logger around responses (many responses include usage—if absent, log just lengths):

// server/lib/usageLog.ts
export function logUsage(label: string, inputTokensApprox: number, outputTokensApprox: number) {
  console.log(JSON.stringify({
    ts: new Date().toISOString(),
    label,
    inputTokensApprox,
    outputTokensApprox
  }));
}


Call it after answers:

const approxIn = prompt.length / 4;     // rough token estimate
const approxOut = answerText.length / 4;
logUsage("rag.answer", approxIn, approxOut);
